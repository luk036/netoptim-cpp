61 template <typename Graph, typename T, typename Fn1, typename Fn2, typename Mapping>
62 auto parametric_search(const Graph& gra, T& r_opt, Fn1&& distrance, Fn2&& zero_cancel,
63 Mapping&& dist, size_t max_iters) {
64 using Edge = typename graph_traits<Graph>::Edge;
65 using Cycle = std::vector<Edge>;
66
67 auto get_weight = [&distrance, &r_opt](const Edge& edge) -> T {
68 return static_cast<T>(distrance(r_opt, edge));
69 };
70
71 auto ncf = NegCycleFinder<Graph>(gra);
72 auto r_min = r_opt;
73 auto c_min = Cycle{};
74 auto c_opt = Cycle{};
75
76 for (auto niter = 0U; niter != max_iters; ++niter) {
77 for (auto&& ci : ncf.howard(dist, get_weight)) {
78 auto ri = static_cast<T>(zero_cancel(ci));
79 if (r_min > ri) {
80 r_min = ri;
81 c_min = std::move(ci);
82 }
83 }
84 if (r_min >= r_opt) break;
85 c_opt = std::move(c_min);
86 r_opt = r_min;
87 }
88 return c_opt;
89 }
