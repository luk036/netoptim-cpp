57 auto assess_optim(const Vec& x, double& gamma) -> std::tuple<Cut, bool> {
58 const auto tcp = x[0];
59 const auto beta = x[1];
60 if (tcp < 0.0) {
61 return {Cut{Vec{-1.0, 0.0}, -tcp}, false};
62 }
63 if (const auto fj = tcp - gamma * beta; fj > 0.0) {
64 return {Cut{Vec{1.0, -gamma}, fj}, false};
65 }
66 const auto get_weight = [tcp, beta](const Edge& edge) -> double {
67 return edge.type == 's' ? tcp - edge.delay - beta : edge.delay - beta;
68 };
69 // Howard may yield spurious non-negative cycles (verification is per-edge),
70 // so keep only the most-violated cycle and cut only if it is negative.
71 auto best = std::optional<Cut>{};
72 auto min_weight = 0.0;
73 for (auto&& C : this->_S.howard(this->_dist, get_weight)) {
74 auto weight_sum = 0.0;
75 auto setup = 0.0;
76 for (auto&& edge : C) {
77 weight_sum += get_weight(edge);
78 setup += edge.type == 's' ? 1.0 : 0.0;
79 }
80 if (weight_sum < min_weight) {
81 min_weight = weight_sum;
82 best = Cut{Vec{-setup, static_cast<double>(C.size())}, -weight_sum};
83 }
84 }
85 if (best) {
86 return {std::move(*best), false};
87 }
88 if (beta == 0.0) {
89 throw std::domain_error("ClockScheduleOracle: beta is zero (unbounded ratio)");
90 }
91 gamma = tcp / beta;
92 return {Cut{Vec{1.0, -1.0}, 0.0}, true};
93 }
